cmake_minimum_required(VERSION 3.25)
project(Scoria VERSION 0.0.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(SCORIA_BUILD_PLUGIN "Build native VST3 and macOS AUv2" ON)
option(IPLUG_DEPLOY_PLUGINS "Install plugins in host search paths" OFF)
include(cmake/Assets.cmake)
include(cmake/Nam.cmake)
add_library(scoria_core STATIC src/models/NamAmplifier.cpp src/models/AssetLoader.cpp src/dsp/WavImpulse.cpp)
set_target_properties(scoria_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
target_link_libraries(scoria_core PUBLIC scoria_nam Threads::Threads)
target_sources(scoria_core INTERFACE $<TARGET_OBJECTS:scoria_nam>)
target_include_directories(scoria_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
include(CTest)
if(BUILD_TESTING)
  add_executable(scoria_render_test tests/RenderDi.cpp)
  target_link_libraries(scoria_render_test PRIVATE scoria_core)
  add_executable(scoria_completion_tests tests/CompletionTests.cpp)
  target_link_libraries(scoria_completion_tests PRIVATE scoria_core)
  add_test(NAME scoria_completion COMMAND scoria_completion_tests "${PROJECT_SOURCE_DIR}/assets")
  add_executable(scoria_tests tests/AudioEngineTests.cpp)
  target_link_libraries(scoria_tests PRIVATE scoria_core)
  add_test(NAME scoria_core COMMAND scoria_tests)
  add_executable(scoria_dsp_tests tests/DspTests.cpp)
  target_link_libraries(scoria_dsp_tests PRIVATE scoria_core)
  add_test(NAME scoria_dsp COMMAND scoria_dsp_tests "${PROJECT_SOURCE_DIR}/assets")
endif()
if(SCORIA_BUILD_PLUGIN)
  if(NOT APPLE AND NOT WIN32)
    message(FATAL_ERROR "Native plugins require macOS or Windows; use -DSCORIA_BUILD_PLUGIN=OFF for core tests")
  endif()
  include(cmake/Dependencies.cmake)
  include("${IPLUG2_DIR}/iPlug2.cmake")
  find_package(iPlug2 REQUIRED)
  add_subdirectory(src/plugin)
endif()
